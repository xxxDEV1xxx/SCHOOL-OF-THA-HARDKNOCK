$adb = $null

    $adb = (Get-Command adb).Source

        "$env:USERPROFILE\AppData\Local\Android\Sdk\platform-tools\adb.exe",
        "C:\Android\platform-tools\adb.exe",
        "C:\Program Files\Android\platform-tools\adb.exe",
        "C:\platform-tools\adb.exe"
    )
    foreach ($c in $candidates) {
        if (Test-Path $c) { $adb = $c; break }
    }
}

if (-not $adb) {
    Write-Fatal "ADB not found. Install Android Platform Tools and add to PATH.`n  https://developer.android.com/tools/releases/platform-tools"
}
Write-OK "ADB: $adb"

# â”€â”€ Step 2: Locate bash (WSL preferred, Git Bash fallback) â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
Write-Step "Locating bash environment..."

$BashMode = $null   # 'wsl' | 'gitbash'
$BashExe  = $null

# 2a. WSL

    $wslCheck = wsl --list --quiet 2>&1
    if ($LASTEXITCODE -eq 0 -and $wslCheck) {
        $BashMode = 'wsl'
        $BashExe  = 'wsl'
        Write-OK "Bash: WSL"
    }
}

# 2b. Git Bash
if (-not $BashMode) {
    $gitBashCandidates = @(
        "C:\Program Files\Git\bin\bash.exe",
        "C:\Program Files (x86)\Git\bin\bash.exe",
        "$env:LOCALAPPDATA\Programs\Git\bin\bash.exe"
    )
    foreach ($g in $gitBashCandidates) {
        if (Test-Path $g) {
            $BashMode = 'gitbash'
            $BashExe  = $g
            Write-OK "Bash: Git Bash ($g)"
            break
        }
    }
}

    Write-Fatal "No bash found. Install WSL (wsl --install) or Git for Windows (https://git-scm.com)."
}

# â”€â”€ Path conversion helper â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
# Converts a Windows path to the form bash expects
function ConvertTo-BashPath {
    param([string]$WinPath)
    switch ($BashMode) {
        'wsl' {
            # wsl.exe --exec wslpath is the safest conversion
            $p = & wsl wslpath -a ($WinPath -replace '\\','/')
            return $p.Trim()
        }
        'gitbash' {
            # Git Bash wants /c/Users/... style
            $p = $WinPath -replace '\\','/'
            if ($p -match '^([A-Za-z]):(.*)') {
                return '/' + $Matches[1].ToLower() + $Matches[2]
            }
            return $p
        }
    }
}

    "deploy_herolte.sh",
    "ondevice_build.sh",
    "aoa_android",

    "sentinel"
)

    if (Test-Path $full) {
        Write-OK "Found: $item"
    } else {
        Write-Warn "MISSING: $item"
        $allPresent = $false
    }
}

    Write-Fatal "One or more required files missing. Ensure deploy_herolte.sh, ondevice_build.sh, aoa_android\, dsl_s_elf\, and sentinel\ are alongside this script."
}

# â”€â”€ Step 4: ADB device check â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
Write-Step "Starting ADB server and checking for device..."

Start-Sleep -Seconds 2

$adbDevices = & $adb devices 2>&1
$deviceLine = $adbDevices | Select-String -Pattern '\bdevice$' | Select-Object -First 1
if (-not $deviceLine) {
    Write-Fatal "No authorised ADB device found.`n  Check: USB cable, USB debugging enabled, trust prompt on phone."
}

$deviceId = ($deviceLine.Line -split '\s+')[0]
Write-OK "Device detected: $deviceId"

# Read device properties
$model  = (& $adb shell getprop ro.product.model 2>&1).Trim() -replace "`r",''

    Write-Warn "Device may not be a herolte (S7). Board='$board'. Proceeding anyway."
}

# â”€â”€ Step 5: Fix line endings (CRLF â†’ LF) â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€

    $path = Join-Path $ScriptDir $sh
    $content = [System.IO.File]::ReadAllText($path)
    $fixed   = $content -replace "`r`n","`n" -replace "`r","`n"
    [System.IO.File]::WriteAllText($path, $fixed, [System.Text.UTF8Encoding]::new($false))
    Write-OK "Fixed: $sh"
}

    'gitbash' { & $BashExe -c "chmod +x '$deployBash' '$ondeviceBash'" }

Write-Host "â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•" -ForegroundColor Cyan
